CREATE SCHEMA IF NOT EXISTS `planteWeb`;
USE `planteWeb`;

CREATE TABLE IF NOT EXISTS `utilisateur` (
  `idUtilisateur` INT NOT NULL AUTO_INCREMENT,
  `nomUtilisateur` VARCHAR(45) NOT NULL UNIQUE,
  `email` VARCHAR(100) NULL,
  `motDePasse` VARCHAR(255) NOT NULL,
  `dateInscription` DATE NOT NULL DEFAULT (CURRENT_DATE),
  PRIMARY KEY (`idUtilisateur`)
);

CREATE TABLE IF NOT EXISTS `plante` (
  `idPlante` INT NOT NULL AUTO_INCREMENT,
  `nom` VARCHAR(45) NOT NULL,
  `typePlante` VARCHAR(100) NULL,
  `dateAcquisition` DATE NOT NULL,
  `utilisateur_idUtilisateur` INT NOT NULL,
  PRIMARY KEY (`idPlante`),
  INDEX `fk_plante_utilisateur1_idx` (`utilisateur_idUtilisateur` ASC),
  CONSTRAINT `fk_plante_utilisateur1`
    FOREIGN KEY (`utilisateur_idUtilisateur`)
    REFERENCES `utilisateur` (`idUtilisateur`)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

CREATE TABLE IF NOT EXISTS `note` (
  `idNote` INT NOT NULL AUTO_INCREMENT,
  `titre` VARCHAR(45) NOT NULL,
  `contenu` TEXT NOT NULL,
  `idPlante` INT NOT NULL,
  PRIMARY KEY (`idNote`),
  INDEX `fk_notes_plante1_idx` (`idPlante` ASC),
  CONSTRAINT `fk_notes_plante1`
    FOREIGN KEY (`idPlante`)
    REFERENCES `plante` (`idPlante`)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);

CREATE TABLE IF NOT EXISTS `typeEvenement` (
  `idTypeEvenement` INT NOT NULL AUTO_INCREMENT,
  `typeEvenement` VARCHAR(45) NOT NULL,
  PRIMARY KEY (`idTypeEvenement`)
);

CREATE TABLE IF NOT EXISTS `evenement` (
  `idEvenement` INT NOT NULL AUTO_INCREMENT,
  `commentaire` TEXT NOT NULL,
  `idPlante` INT NOT NULL,
  `idTypeEvenement` INT NOT NULL,
  `date` DATE NOT NULL,
  PRIMARY KEY (`idEvenement`),
  INDEX `fk_evenement_plante1_idx` (`idPlante` ASC),
  INDEX `fk_evenement_typeEvenement1_idx` (`idTypeEvenement` ASC),
  CONSTRAINT `fk_evenement_plante1`
    FOREIGN KEY (`idPlante`)
    REFERENCES `plante` (`idPlante`)
    ON DELETE CASCADE
    ON UPDATE CASCADE,
  CONSTRAINT `fk_evenement_typeEvenement1`
    FOREIGN KEY (`idTypeEvenement`)
    REFERENCES `typeEvenement` (`idTypeEvenement`)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);
